class Game(tk.Tk):
    def __init__(self):
        super().__init__()
        self.title("六一零食大闯关")
        self.geometry("800x600")

        self.canvas.pack(fill="both", expand=True)

        self.setup_game()
        
        self.bind("<KeyPress>", self.on_key)
        self.bind("<KeyRelease>", self.off_key)
        
        self.game_loop()
        
    def setup_game(self):
        # 创建玩家
        self.player = self.world.create_entity()
        self.world.add_component(self.player, Transform(400, 300))
        self.world.add_component(self.player, Render(text="👄"))
        self.world.add_component(self.player, Movement(speed=8))
        self.world.add_component(self.player, Health())
        
        # 创建零食（随机撒满屏幕）
        import random
        from entities import SNACK_LIST
        for _ in range(50):
            create_snack(self.world, random.randint(0, 780), random.randint(0, 580), random.choice(SNACK_LIST))
            
    def on_key(self, e):
        move = self.world.component_for_entity(self.player, Movement)
        if e.keysym == 'Left': move.vx = -1
        if e.keysym == 'Right': move.vx = 1
        if e.keysym == 'Up': move.vy = -1
        if e.keysym == 'Down': move.vy = 1
        
    def off_key(self, e):
        move = self.world.component_for_entity(self.player, Movement)
        move.vx = 0
        move.vy = 0

    def game_loop(self):
        # 更新系统
        self.world.update(MovementSystem())
        self.world.update(RenderSystem(self.canvas))
        
        self.after(30, self.game_loop) # 30ms刷新一次

if __name__ == "__main__":

# ecs.py
import uuid

class World:
    """
    ECS 世界：负责存储和管理所有实体（Entity）、组件（Component）和系统（System）。
    """
    def __init__(self):
        # 存储格式: {entity_id: {ComponentType: component_instance}}
        self._entities = {}
        # 存储待删除的实体，避免在迭代中修改字典
        self._dead_entities = set()

    def create_entity(self):
        """创建一个新的实体（Entity），返回一个唯一的ID。"""
        entity_id = str(uuid.uuid4())[:8]  # 短ID，好看
        self._entities[entity_id] = {}
        return entity_id

    def add_component(self, entity, component_instance):
        """给实体添加一个组件。"""
        comp_type = type(component_instance)
        if entity in self._entities:
            self._entities[entity][comp_type] = component_instance

    def remove_component(self, entity, component_type):
        """移除实体的某个组件。"""
        if entity in self._entities and component_type in self._entities[entity]:
            del self._entities[entity][component_type]

    def component_for_entity(self, entity, component_type):
        """获取实体特定的组件实例（用于单例访问，如Player）。"""
        if entity in self._entities:
            return self._entities[entity].get(component_type)
        return None

    def get_components(self, *component_types):
        """
        生成器：返回所有拥有指定组件组合的实体及其组件实例。
        这是 System 获取数据的主要方式。
        用法: for ent, (trans, rend) in world.get_components(Transform, Render):
        """
        for entity, components in self._entities.items():
            if all(comp_type in components for comp_type in component_types):
                yield entity, [components[ct] for ct in component_types]

    def delete_entity(self, entity):
        """标记实体为待删除状态。"""
        if entity in self._entities:
            self._dead_entities.add(entity)

    def clear_dead_entities(self):
        """清理已死亡的实体（通常在帧末调用）。"""
        for entity in self._dead_entities:
            if entity in self._entities:
                del self._entities[entity]
        self._dead_entities.clear()

    def update(self, system_instance, *args, **kwargs):
        """
        执行一个系统（System）。
        系统通过调用 world.get_components() 来获取需要处理的数据。
        """
        system_instance.update(self, *args, **kwargs)

💥 CollisionSystem（碰撞系统）

这个系统会遍历所有带有 Transform（位置）和 Collider（碰撞体）组件的实体，检查是否重叠。

# systems.py (追加这个类)

class Collider:
    """新增一个组件：标记该实体具有碰撞体积"""
    def __init__(self, radius=15):
        self.radius = radius

class CollisionSystem:
    """
    处理所有碰撞逻辑。
    规则：
    1. 玩家碰到大白兔/AD钙 -> 加分 + 回血
    2. 玩家碰到魔鬼辣片 -> 扣血
    3. 碰到后零食消失
    """
    def __init__(self, player_entity_id):
        self.player_id = player_entity_id

    def update(self, world):
        player_trans = world.component_for_entity(self.player_id, Transform)
        player_collider = world.component_for_entity(self.player_id, Collider)
        player_health = world.component_for_entity(self.player_id, Health)
        player_score = world.component_for_entity(self.player_id, Score)

        # 获取所有零食实体 (有Transform, Render, Collider，但不是玩家)
        for snack_id, (snack_trans, snack_render, snack_collider) in world.get_components(Transform, Render, Collider):
            
            if snack_id == self.player_id:
                continue

            # 简单的圆形碰撞检测 (距离 < 半径之和)
            dx = player_trans.x - snack_trans.x
            dy = player_trans.y - snack_trans.y
            distance = (dx*dx + dy*dy) ** 0.5

            if distance < (player_collider.radius + snack_collider.radius):
                # 发生碰撞！
                print(f"吃到了: {snack_render.text}")
                
                # --- 判定零食类型 (基于 Render.text 或 Color) ---
                # 你的“抢劫式宣言”里的分类
                healing_items = ["🐰", "🍼", "🍬"]  # 大白兔, AD钙, 糖
                damaging_items = ["🌶️"]             # 魔鬼辣片
                
                if snack_render.text in healing_items:
                    player_health.hp += 10
                    player_score.value += 100
                    print(f"回血! HP: {player_health.hp}")
                
                elif snack_render.text in damaging_items:
                    player_health.hp -= 20
                    print(f"好辣! HP: {player_health.hp}")

                # 销毁零食 (标记为删除)
                world.delete_entity(snack_id)

5. 修改 RenderSystem（显示 Game Over）

在屏幕上画出巨大的 “GAME OVER”。

# systems.py (修改 RenderSystem)

class RenderSystem:
    def __init__(self, canvas: tk.Canvas):
        self.canvas = canvas
        
    def update(self, world):
        self.canvas.delete("all")
        
        # 绘制实体
        for ent, (trans, render) in world.get_components(Transform, Render):
            self.canvas.create_text(trans.x, trans.y, text=render.text, font=("Segoe UI Emoji", trans.size))

        # 绘制UI
        player_id = list(world.get_components(GameState))[0][0]
        health = world.component_for_entity(player_id, Health)
        score = world.component_for_entity(player_id, Score)
        state = world.component_for_entity(player_id, GameState)

        self.canvas.create_text(10, 10, text=f"HP: {max(0, health.hp)}", fill="white", font=("Arial", 16), anchor="nw")
        self.canvas.create_text(10, 40, text=f"Score: {score.value}", fill="yellow", font=("Arial", 16), anchor="nw")

        # 游戏结束画面
        if state.is_over:
            self.canvas.create_rectangle(0, 200, 800, 400, fill="black", outline="red", width=5)
            self.canvas.create_text(400, 280, text="GAME OVER", fill="red", font=("Arial Black", 60))
            self.canvas.create_text(400, 350, text="Press SPACE to Restart", fill="white", font=("Arial", 20))
