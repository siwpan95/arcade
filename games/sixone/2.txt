# data_loader.py
import pandas as pd
from components import SnackData

class SnackConfig:
    def __init__(self, file_path="snacks.xlsx"):
        self.df = pd.read_excel(file_path)
        # 将 DataFrame 转换为字典，方便通过 ID 查找
        self.config_map = self.df.set_index('id').T.to_dict()

    def get_snack_by_id(self, snack_id):
        """根据ID获取零食配置"""
        if snack_id in self.config_map:
            cfg = self.config_map[snack_id]
            return SnackData(
                cfg['name'], cfg['icon'], cfg['type'], 
                cfg['hp_change'], cfg['score'], cfg['color']
            )
        return None

    def get_random_snack_id(self):
        """随机获取一个零食ID"""
        return self.df.sample(1).iloc[0]['id']

🎮 第三步：修改碰撞系统（逻辑分发）

CollisionSystem 现在变成一个分发器，根据 SnackData.type 调用不同的处理逻辑。

# systems.py (修改 CollisionSystem)

import time

class CollisionSystem:
    def __init__(self, player_entity_id):
        self.player_id = player_entity_id

    def update(self, world):
        state = world.component_for_entity(self.player_id, GameState)
        if state.is_over:
            return

        player_trans = world.component_for_entity(self.player_id, Transform)
        player_collider = world.component_for_entity(self.player_id, Collider)
        player_health = world.component_for_entity(self.player_id, Health)
        player_score = world.component_for_entity(self.player_id, Score)
        player_status = world.component_for_entity(self.player_id, StatusEffect)
        
        current_time = time.time() * 1000

        for snack_id, (snack_trans, snack_data, snack_collider) in world.get_components(Transform, SnackData, Collider):
            if snack_id == self.player_id:
                continue

            # 无敌状态不触发碰撞
            if player_status.is_invincible:
                continue

            dx = player_trans.x - snack_trans.x
            dy = player_trans.y - snack_trans.y
            distance = (dx*dx + dy*dy) ** 0.5

            if distance < (player_collider.radius + snack_collider.radius):
                # 基础数值变化
                player_health.hp += snack_data.hp_change
                player_score.value += snack_data.score_value

                # 根据 Type 触发特殊逻辑
                if snack_data.type == "heal":
                    pass # 普通回血
                
                elif snack_data.type == "damage":
                    pass # 普通掉血
                
                elif snack_data.type == "buff":
                    # 泡泡糖：获得护盾
                    if not player_status.is_invincible:
                        player_status.is_invincible = True
                        player_status.effect_end_time = current_time + snack_data.duration
                        world.add_component(self.player_id, Shield(snack_data.value))
                        print("获得护盾！")
                
                elif snack_data.type == "debuff":
                    # 秀逗糖：反向控制
                    if not player_status.is_reversed:
                        player_status.is_reversed = True
                        player_status.effect_end_time = current_time + snack_data.duration
                        print("反向移动！")
                
                elif snack_data.type == "special":
                    # 钻石糖：清屏
                    self.clear_screen(world)
                    print("清屏！")

                world.delete_entity(snack_id)

        # 状态过期检测
        if player_status.is_invincible or player_status.is_reversed:
            if current_time > player_status.effect_end_time:
                player_status.is_invincible = False
                player_status.is_reversed = False
                if world.component_for_entity(self.player_id, Shield):
                    world.remove_component(self.player_id, Shield)
                print("状态恢复")

    def clear_screen(self, world):
        """删除所有零食"""
        for ent, (snack_data,) in world.get_components(SnackData):
            if ent != self.player_id:
                world.delete_entity(ent)

class MovementSystem:
    def update(self, world, dt):
        for ent, (trans, move, status) in world.get_components(Transform, Movement, StatusEffect):
            vx = move.vx
            vy = move.vy
            
            # 如果是反向状态，反转方向
            if status.is_reversed:
                vx *= -1
                vy *= -1
                
            trans.x += vx * move.speed
            trans.y += vy * move.speed
